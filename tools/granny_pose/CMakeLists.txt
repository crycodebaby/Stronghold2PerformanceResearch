cmake_minimum_required(VERSION 3.20)
project(granny_pose_contract LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "This host requires MSVC Win32, matching the verified DLL ABI")
endif()
add_executable(pose_contract_host pose_contract_host.cpp pose_math.cpp)
target_compile_features(pose_contract_host PRIVATE cxx_std_17)
target_compile_options(pose_contract_host PRIVATE /W4 /WX /fp:strict)
target_link_libraries(pose_contract_host PRIVATE bcrypt)
